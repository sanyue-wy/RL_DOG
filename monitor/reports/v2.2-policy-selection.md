v2.1版本中，Sim2Real验证功能的策略路径是硬编码的（`model_1200.pt`），用户无法：
- 选择不同的训练checkpoint进行验证

- 快速切换验证最新训练的策略

实现完整的策略文件扫描和选择功能：
- ✅ 自动扫描logs目录下所有训练策略文件
- ✅ 分组显示（按实验类型和训练运行）
- ✅ 下拉菜单选择任意checkpoint
- ✅ 命令行参数传递策略路径

**get_available_policies()**
```python
def get_available_policies(self):
    """扫描logs目录，返回所有可用的策略文件"""
    policies = []
    logs_dir = self.project_root / "legged_gym" / "logs"

    if not logs_dir.exists():
        return policies

    # 遍历实验目录（blindplane, blindrough3等）
    for exp_dir in logs_dir.iterdir():
        if not exp_dir.is_dir():
            continue

        # 遍历训练运行目录（Jul31_14-43-25_等）
        for run_dir in exp_dir.iterdir():
            if not run_dir.is_dir():
                continue

            # 扫描模型文件（model_0.pt, model_100.pt等）
            model_files = list(run_dir.glob("model_*.pt"))

            for model_file in model_files:
                try:
                    iteration = int(model_file.stem.split('_')[1])
                    policies.append({
                        'path': str(model_file),
                        'experiment': exp_dir.name,
                        'run': run_dir.name,
                        'iteration': iteration,
                        'filename': model_file.name,
                        'display_name': f"{exp_dir.name}/{run_dir.name}/{model_file.name}"
                    })
                except (ValueError, IndexError):
                    continue

    # 排序：先按实验名，再按运行时间，最后按迭代次数
    policies.sort(key=lambda x: (x['experiment'], x['run'], x['iteration']))

    return policies
```

**find_latest_policy()**
```python
def find_latest_policy(self):
    """查找最新的策略文件（按修改时间）"""
    policies = self.get_available_policies()
    if not policies:
        return None

    # 按修改时间排序，返回最新的
    policies_with_mtime = []
    for p in policies:
        try:
            mtime = Path(p['path']).stat().st_mtime
            policies_with_mtime.append((p, mtime))
        except:
            continue

    if policies_with_mtime:
        policies_with_mtime.sort(key=lambda x: x[1], reverse=True)
        return policies_with_mtime[0][0]['path']

    return policies[-1]['path'] if policies else None
```

**start_sim2real() 修改**
```python
def start_sim2real(self, policy_path=None):
    """启动 Sim2Real 验证（支持指定策略路径）"""
    if self.sim2real_running:
        return False, "Sim2Real 验证已在运行中"

    # 如果没有指定策略，自动查找最新
    if not policy_path:
        policy_path = self.find_latest_policy()
        if not policy_path:
            return False, "没有找到可用的策略文件"

    # 验证策略文件存在
    if not Path(policy_path).exists():
        return False, f"策略文件不存在: {policy_path}"

    sim2sim_path = self.project_root / "legged_gym" / "sim2sim.py"

    # 通过命令行参数传递策略路径
    cmd = [
        sys.executable,
        "-u",
        str(sim2sim_path),
        "--policy", str(policy_path)
    ]

    try:
        self.sim2real_process = subprocess.Popen(
            cmd,
            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,
            universal_newlines=True,
            bufsize=0,
            cwd=str(self.project_root / "legged_gym"),
            env={**os.environ, 'PYTHONUNBUFFERED': '1'}
        )
        self.sim2real_running = True

        # 启动输出读取线程
        thread = threading.Thread(target=self._read_sim2real_output, daemon=True)
        thread.start()

        return True, f"Sim2Real 验证已启动: {Path(policy_path).name}"
    except Exception as e:
        return False, f"Sim2Real 启动失败: {str(e)}"
```

@sio.on('get_policies')
async def handle_get_policies(sid):
    """获取可用策略列表"""
    print(f"[Socket.IO] get_policies received")
    try:
        policies = train_manager.get_available_policies()
        await sio.emit('policies_list', {'policies': policies}, to=sid)
    except Exception as e:
        print(f"[Socket.IO] get_policies error: {e}")
        await sio.emit('policies_list', {'policies': [], 'error': str(e)}, to=sid)

// 按实验和运行分组策略
const groupedPolicies = useMemo(() => {
  const groups = {}
  policies.forEach(policy => {
    const expKey = policy.experiment
    if (!groups[expKey]) {
      groups[expKey] = {}
    }
    const runKey = policy.run
    if (!groups[expKey][runKey]) {
      groups[expKey][runKey] = []
    }
    groups[expKey][runKey].push(policy)
  })
  return groups
}, [policies])
```

def find_latest_model():
    """自动查找最新的模型文件（按修改时间）"""
    logs_dir = Path(_SCRIPT_DIR) / "logs"
    all_models = []

    for model_file in logs_dir.rglob("model_*.pt"):
        try:
            iteration = int(model_file.stem.split('_')[1])
            all_models.append((model_file, iteration, model_file.stat().st_mtime))
        except (ValueError, IndexError):
            continue

    if not all_models:
        return None

    # 按修改时间排序，返回最新的
    all_models.sort(key=lambda x: x[2], reverse=True)
    return str(all_models[0][0])
```

def main():
    # 解析命令行参数
    parser = argparse.ArgumentParser(description='Sim2Sim MuJoCo验证')
    parser.add_argument('--policy', type=str, default=None,

## 🔧 V2.2.1 - 模型维度自动检测修复

### 问题描述

v2.2版本发布后发现两个严重问题：

1. **观测维度不匹配**：不同训练运行使用不同的观测配置
   - `blindplane` 训练：`height_command=True` → 46维/步 × 6历史 = 276维
   - `blindrough3` 训练：`height_command=False` → 45维/步 × 6历史 = 270维
   - sim2sim.py 硬编码为46维，导致270维模型报错：`RuntimeError: mat1 and mat2 shapes cannot be multiplied (1x276 and 270x128)`

2. **Socket.IO事件不发送**：`_emit_sync()` 依赖 `self.loop`，但 `start_sim2real()` 没有初始化事件循环
   - 导致前端收不到 `sim2real_output` 事件
   - 只有 `sim2real_status` 事件能收到（因为是直接 await 发送）

### 解决方案

#### 1. 模型维度自动检测

修改 `load_policy()` 函数，从模型权重自动推断观测配置：

```python

    ckpt = torch.load(path, map_location=device, weights_only=False)

    model_config = {
        'num_one_step_obs': num_one_step_obs,
        'num_obs': est_input_dim,
        'has_height_cmd': has_height_cmd,
        'commands_scale': np.array([2.0, 2.0, 0.25, 4.0] if has_height_cmd else [2.0, 2.0, 0.25]),

    }

```
